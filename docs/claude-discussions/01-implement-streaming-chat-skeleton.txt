     Implementation sketch

     import asyncio
     import os

     from anthropic import AsyncAnthropic
     from dotenv import load_dotenv

     load_dotenv()

     MODEL = "claude-sonnet-4-6"
     MAX_TOKENS = 1024
     SYSTEM_PROMPT = "You are a helpful CLI assistant. Keep replies concise."

     client = AsyncAnthropic(api_key=os.environ["ANTHROPIC_API_KEY"])


     async def get_user_input() -> str:
         # input() blocks the event loop; offload it so async work can run 
     alongside.
         return await asyncio.to_thread(input, "You: ")


     async def call_llm(user_input: str, conversation_history: list):
         conversation_history.append({"role": "user", "content": user_input})

         assistant_text_parts: list[str] = []
         async with client.messages.stream(
             model=MODEL,
             max_tokens=MAX_TOKENS,
             system=SYSTEM_PROMPT,
             messages=conversation_history,
         ) as stream:
             async for delta in stream.text_stream:
                 assistant_text_parts.append(delta)
                 yield delta

         conversation_history.append(
             {"role": "assistant", "content": "".join(assistant_text_parts)}
         )


     # get_weather / research_topic stubs unchanged for now.


     async def main():
         conversation_history: list = []
         print("Chat ready. Type 'quit' to exit.")
         while True:
             user_input = (await get_user_input()).strip()
             if not user_input:
                 continue
             if user_input.lower() in {"quit", "exit", "q"}:
                 break

             print("Assistant: ", end="", flush=True)
             async for chunk in call_llm(user_input, conversation_history):
                 print(chunk, end="", flush=True)
             print()


     if __name__ == "__main__":
         asyncio.run(main())

     Notes on the design

     - AsyncAnthropic + messages.stream() — the SDK's async streaming context
     manager yields text deltas via stream.text_stream. This is the idiomatic
     streaming path and avoids hand-parsing event types. When we add tools later
      we can switch to iterating raw events (async for event in stream) to react
      to content_block_start / tool_use blocks; for a text-only skeleton,
     text_stream is the simpler call.
     - History shape — Anthropic's messages API uses {"role": 
     "user"|"assistant", "content": ...} and system is a top-level param, not a
     message. We append the user message before the call and the assistant
     message after the stream completes so a mid-stream crash doesn't leave a
     half-recorded turn.
     - asyncio.to_thread(input, ...) — input() is synchronous and would block
     the event loop. Offloading it keeps the loop free, which matters later when
      we want a spinner running during slow tool calls.
     - Skip empty input — small UX polish; avoids sending blank turns to the
     API.
